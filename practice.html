        function loadFlexLevel(lvl) {
            currentFlexLevel = lvl;
            document.querySelectorAll('.level-btn').forEach((btn, idx) => {
                btn.classList.toggle('active', idx + 1 === lvl);
            });

            const data = flexLevels[lvl];
            document.getElementById('flex-level-title').innerText = data.title;
            document.getElementById('flex-level-desc').innerText = data.desc;

            const targetLayer = document.getElementById('target-layer');
            targetLayer.style.cssText = data.targetLayerStyle;

            // reset
            document.getElementById('jc-select').value = "flex-start";
            document.getElementById('ai-select').value = "flex-start";

            const itemsLayer = document.getElementById('items-layer');
            itemsLayer.style.justifyContent = "flex-start";
            itemsLayer.style.alignItems = "flex-start";

            document.querySelectorAll('.frog-item').forEach(f => f.classList.remove('success'));
            const banner = document.getElementById('flex-status-banner');
            banner.style.display = 'none';
        }

        function submitFlexGame() {
            const jc = document.getElementById('jc-select').value;
            const ai = document.getElementById('ai-select').value;

            const itemsLayer = document.getElementById('items-layer');
            itemsLayer.style.justifyContent = jc;
            itemsLayer.style.alignItems = ai;

            const target = flexLevels[currentFlexLevel];
            const banner = document.getElementById('flex-status-banner');

            if (jc === target.targetJc && ai === target.targetAi) {
                document.querySelectorAll('.frog-item').forEach(f => f.classList.add('success'));
                banner.className = 'status-banner success';
                banner.innerHTML = `<i class="fa-solid fa-circle-check"></i> Perfect! Frogs landed on the targets! Level Passed!`;
                banner.style.display = 'flex';
            } else {
                document.querySelectorAll('.frog-item').forEach(f => f.classList.remove('success'));
                banner.className = 'status-banner error';
                banner.innerHTML = `<i class="fa-solid fa-circle-xmark"></i> Not aligned! Target needs justify-content: ${target.targetJc} and align-items: ${target.targetAi}.`;
                banner.style.display = 'flex';
            }
        }

        loadFlexLevel(1);
