Question 3: HTML5 Form Validation, Datalist & Attributes

HTML5 Forms

Advanced form validation showcasing semantic <fieldset> grouping, university ID regex pattern constraints, datalist suggestions, number stepping, and readonly/disabled field controls.

👁 Click Output Preview to Inspect Code in Draggable Split View
📄 Question 3 Source & Live Output
SOURCE CODE (HTML5) 
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Registration & Validation Form</title>
</head>
<body>

    <form action="/register" method="post">
        <fieldset>
            <legend>Form to fill</legend>

            <label for="username">Username:</label>
            <input type="email" id="username" name="username" required autocomplete="email" autofocus>

            <label for="reg">REGISTER:</label>
            <input name="reg" id="reg" pattern="FA[0-9]{2}-BSSE-[0-9]{3}" required>
            <br>

            <label for="password">Password:</label>
            <input type="password" id="password" name="password" required>

            <label for="age">Age:</label>
            <input type="number" id="age" name="age" min="10" max="20" step="5" required>
            <br>

            <label for="cities">Cities:</label>        
            <input list="cities" id="cities-input" name="city">
            <datalist id="cities">
                <option value="vehari">
                <option value="multan">
                <option value="islamabad">
            </datalist>

            <label for="phone">Phone:</label>
            <input type="tel" id="phone" name="phone" pattern="\+92-[0-9]{11}" placeholder="+9234344444444" required readonly>
            <br>

            <label for="search">SEARCH:</label>
            <input type="search" id="search" name="search" required>
            <br>

            <label for="url">Search website here:</label>
            <input type="url" id="url" name="url" disabled>

            <label for="gender">GENDER:</label>
            <input type="radio" id="male" name="gender" value="male" required>
            <label for="male">Male</label>
            <input type="radio" id="female" name="gender" value="female" required>
            <label for="female">Female</label>
            <br>

            <select name="country" id="country" required>
                <option value="">Select a country</option>
                <option value="pakistan">Pakistan</option>
                <option value="usa">USA</option>
                <option value="canada">Canada</option>
                <option value="uk">UK</option>
            </select>

            <label for="date">DATE:</label>
            <input type="date" id="date" name="date" required>

            <label for="address">Address:</label>
            <input type="text" id="address" name="address" required>

            <button type="submit">Register</button>
        </fieldset>
    </form>

</body>
</html>
